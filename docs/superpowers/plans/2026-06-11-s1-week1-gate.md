- [x] **Step 3:【Claude】写 `spikes/datajuicer_ray/wds_to_jsonl.py`**(tar 解包 → `{"text":…,"images":[…]}` jsonl;Spike 2 与后续管线共用)

```python
# /// script
# requires-python = ">=3.12"
# dependencies = []
# ///
"""webdataset tar(.tar 内 <key>.jpg + <key>.txt)→ Data-Juicer 多模态 jsonl。
用法:python wds_to_jsonl.py <tar_dir> <out_dir>  → out_dir/images/* + out_dir/data.jsonl"""
import json, sys, tarfile
from pathlib import Path

def main(tar_dir: str, out_dir: str):
    out = Path(out_dir); img_dir = out / "images"; img_dir.mkdir(parents=True, exist_ok=True)
    n = 0
    with open(out / "data.jsonl", "w") as fh:
        for tar_path in sorted(Path(tar_dir).glob("*.tar")):
            with tarfile.open(tar_path) as tf:
                texts, imgs = {}, {}
                for m in tf.getmembers():
                    key, dot, ext = m.name.rpartition(".")
                    if ext == "txt":
                        texts[key] = tf.extractfile(m).read().decode("utf-8", "ignore").strip()
                    elif ext in ("jpg", "jpeg", "png", "webp"):
                        imgs[key] = (m, ext)
                for key, (m, ext) in imgs.items():
                    if key not in texts:
                        continue
                    p = img_dir / f"{tar_path.stem}_{key.replace('/', '_')}.{ext}"
                    p.write_bytes(tf.extractfile(m).read())
                    fh.write(json.dumps({"text": texts[key], "images": [str(p.resolve())]}) + "\n")
                    n += 1
    print(f"wrote {n} samples → {out/'data.jsonl'}")

if __name__ == "__main__":
    main(sys.argv[1], sys.argv[2])
```
